cmake_minimum_required(VERSION 3.21)

project(workload-network-contract
        VERSION 1.0.0
        DESCRIPTION "Vendor-neutral C++20 runtime for workload network contracts"
        LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "In-source builds are not allowed. Configure into a separate build directory.")
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
endif()

option(WNC_BUILD_TOOLS "Build CLI and coordinator tooling" ON)
option(WNC_BUILD_TESTS "Build the test suites" ON)
option(WNC_BUILD_EXAMPLES "Build the examples" ON)
option(WNC_BUILD_SHARED "Build the core library as a shared library" OFF)
option(WNC_SANITIZER "Instrument with AddressSanitizer and UndefinedBehaviorSanitizer where supported" OFF)
option(WNC_HARDENING "Enable stack protection and control-flow guard where supported" ON)
option(WNC_STRICT_WARNINGS "Treat first-party warnings as errors" ON)

if(MSVC)
  # NOMINMAX and WIN32_LEAN_AND_MEAN keep the Windows headers from defining
  # min/max macros and from pulling in unrelated subsystems.
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  if(WNC_SANITIZER)
    # The address sanitizer runtime requires the dynamic C runtime.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDebugDLL" CACHE STRING "" FORCE)
  endif()
endif()

include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(BuildOptions)

# The version is published to consumers as a generated header, so a downstream
# build can assert which boundary it compiled against without parsing CMake.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/wnc/version.hpp"
"// Workload Network Contract ${PROJECT_VERSION}
// Generated by CMake from the project version. Do not edit.
#ifndef WNC_VERSION_HPP
#define WNC_VERSION_HPP

#define WNC_VERSION_MAJOR ${PROJECT_VERSION_MAJOR}
#define WNC_VERSION_MINOR ${PROJECT_VERSION_MINOR}
#define WNC_VERSION_PATCH ${PROJECT_VERSION_PATCH}
#define WNC_VERSION_STRING \"${PROJECT_VERSION}\"

#endif  // WNC_VERSION_HPP
")

add_subdirectory(src)
if(WNC_BUILD_TOOLS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/CMakeLists.txt")
  add_subdirectory(tools)
endif()
if(WNC_BUILD_EXAMPLES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/CMakeLists.txt")
  add_subdirectory(examples)
endif()
if(WNC_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

install(EXPORT workload-network-contract-targets
        NAMESPACE WorkloadNetworkContract::
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/workload-network-contract
        FILE workload-network-contract-targets.cmake)

configure_package_config_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/cmake/workload-network-contract-config.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/workload-network-contract-config.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/workload-network-contract)

write_basic_package_version_file(
  "${CMAKE_CURRENT_BINARY_DIR}/workload-network-contract-config-version.cmake"
  VERSION ${PROJECT_VERSION}
  COMPATIBILITY SameMajorVersion
  ARCH_INDEPENDENT)

install(FILES
  "${CMAKE_CURRENT_BINARY_DIR}/workload-network-contract-config.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/workload-network-contract-config-version.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/workload-network-contract)

install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
              "${CMAKE_CURRENT_SOURCE_DIR}/README.md"
        DESTINATION ${CMAKE_INSTALL_DOCDIR})
